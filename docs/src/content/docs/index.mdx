import { Card, CardGrid } from "@astrojs/starlight/components";

```esquadro
package catalog

/// Something the store sells.
entity Product {
  name     string  required min(2) max(120)
  price    decimal required min(0)
  category > Category required
  tags     <> Tag
}
```

<CardGrid>
  <Card title="A model, not boilerplate" icon="pencil">
    Entities, fields, rules and relations in `.esq` files; the code, the
    migrations and the API contract follow from them.
  </Card>
  <Card title="Code you would write" icon="seti:go">
    Modules by domain, one file per use case, no runtime magic. Generated code
    passes `go vet` and `gopls` clean.
  </Card>
  <Card title="Regenerate safely" icon="random">
    Change the model and run `esquadro gen` again: new migrations are written,
    and your own edits are never overwritten.
  </Card>
  <Card title="Blueprints" icon="puzzle">
    Start from an API, a worker or a command line tool, then `esquadro add`
    what you need: Postgres, migrations, a JSON API, background work, an
    image... or write your own, with a test.
  </Card>
</CardGrid>
